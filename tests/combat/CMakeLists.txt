set(path "${CMAKE_CURRENT_SOURCE_DIR}/../../engine/code/game/g_combat.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
file(READ "${path}" source)
set(code "")
foreach(marker IN ITEMS "#define\tSWORD_PARRY_KICK" "#define\tSWORD_RIPOSTE_LOCK")
 string(FIND "${source}" "${marker}" start)
 string(SUBSTRING "${source}" ${start} -1 tail)
 string(FIND "${tail}" "\n" length)
 string(SUBSTRING "${tail}" 0 ${length} line)
 string(APPEND code "${line}\n")
endforeach()
foreach(marker IN ITEMS "int CheckArmor (" "static float G_SpeedDamageScale(" "void G_Damage(")
 string(FIND "${source}" "${marker}" start)
 string(SUBSTRING "${source}" ${start} -1 tail)
 string(FIND "${tail}" "\n}" length)
 math(EXPR length "${length}+2")
 string(SUBSTRING "${tail}" 0 ${length} func)
 string(APPEND code "${func}\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/combat_under_test.h" "${code}")
add_executable(combat_regression combat_regression.c ../../engine/code/game/bg_misc.c ../../engine/code/qcommon/q_math.c ../../engine/code/qcommon/q_shared.c)
target_include_directories(combat_regression PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(combat_regression PRIVATE _CRT_SECURE_NO_WARNINGS)
if(NOT MSVC)
 target_link_libraries(combat_regression PRIVATE m)
endif()
foreach(case IN ITEMS no-armor clean-parry raise-boundary raise-early rear-hit glancing wrong-guard guard-break armor armor-bypass friendly-fire lethal armor-limited guard-break-boundary guard-break-expired protection-bypass knockback)
 add_test(NAME combat.${case} COMMAND combat_regression ${case})
endforeach()
