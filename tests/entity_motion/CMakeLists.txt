cmake_minimum_required(VERSION 3.25)
project(entity_motion_regression C)
set(extracted "")
foreach(part IN ITEMS motion reset)
  if(part STREQUAL "motion")
    set(path "${CMAKE_CURRENT_SOURCE_DIR}/../../engine/code/cgame/cg_ents.c")
    set(marker "static void CG_SmoothEntityMotion(")
  else()
    set(path "${CMAKE_CURRENT_SOURCE_DIR}/../../engine/code/cgame/cg_players.c")
    set(marker "void CG_ResetPlayerEntity(")
  endif()
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
  file(READ "${path}" source)
  string(FIND "${source}" "${marker}" begin)
  if(begin LESS 0)
    message(FATAL_ERROR "Entity motion function not found")
  endif()
  string(SUBSTRING "${source}" ${begin} -1 tail)
  string(FIND "${tail}" "\n}" end)
  if(end LESS 0)
    message(FATAL_ERROR "Entity motion function end not found")
  endif()
  math(EXPR length "${end} + 2")
  string(SUBSTRING "${tail}" 0 ${length} body)
  string(APPEND extracted "${body}\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/motion_under_test.h" "${extracted}")
add_executable(entity_motion_regression entity_motion_regression.c
  ../../engine/code/qcommon/q_shared.c ../../engine/code/qcommon/q_math.c)
target_compile_definitions(entity_motion_regression PRIVATE _CRT_SECURE_NO_WARNINGS)
target_include_directories(entity_motion_regression PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
foreach(case IN ITEMS normal-to-slow reset-nearby stale-history steady-motion predicted-player)
  add_test(NAME motion.${case} COMMAND entity_motion_regression ${case})
endforeach()
