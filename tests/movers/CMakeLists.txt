set(path "${CMAKE_CURRENT_SOURCE_DIR}/../../engine/code/game/g_mover.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
file(READ "${path}" source)
string(FIND "${source}" "typedef struct {" start)
string(FIND "${source}" "qboolean\tG_TryPushingEntity(" func)
string(SUBSTRING "${source}" ${func} -1 tail)
string(FIND "${tail}" "\n}" finish)
math(EXPR length "${func}-${start}+${finish}+2")
string(SUBSTRING "${source}" ${start} ${length} code)
foreach(marker IN ITEMS "qboolean G_MoverPush(" "void G_MoverTeam(")
 string(FIND "${source}" "${marker}" start)
 string(SUBSTRING "${source}" ${start} -1 tail)
 string(FIND "${tail}" "\n}" length)
 math(EXPR length "${length}+2")
 string(SUBSTRING "${tail}" 0 ${length} func)
 string(APPEND code "\n${func}\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/mover_under_test.h" "${code}")
add_executable(mover_regression mover_regression.c ../../engine/code/game/bg_misc.c ../../engine/code/qcommon/q_math.c ../../engine/code/qcommon/q_shared.c)
target_include_directories(mover_regression PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(mover_regression PRIVATE _CRT_SECURE_NO_WARNINGS)
if(NOT MSVC)
 target_link_libraries(mover_regression PRIVATE m)
endif()
foreach(case IN ITEMS translation rotation blocked-rotation stop-nonrider stop-rider multi-carry multi-rollback)
 add_test(NAME mover.${case} COMMAND mover_regression ${case})
endforeach()
