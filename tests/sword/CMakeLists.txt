set(weapon_source "${CMAKE_CURRENT_SOURCE_DIR}/../../engine/code/game/g_weapon.c")
set(move_source "${CMAKE_CURRENT_SOURCE_DIR}/../../engine/code/game/bg_pmove.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${weapon_source}" "${move_source}")
file(READ "${weapon_source}" source)
string(FIND "${source}" "#define\tSWORD_RANGE" begin)
string(FIND "${source}" "void Weapon_Sword(" func)
string(SUBSTRING "${source}" ${func} -1 tail)
string(FIND "${tail}" "\n}" finish)
math(EXPR length "${func} - ${begin} + ${finish} + 2")
string(SUBSTRING "${source}" ${begin} ${length} code)
file(READ "${move_source}" source)
string(FIND "${source}" "static float PM_SwordMagnetLunge(" begin)
string(SUBSTRING "${source}" ${begin} -1 tail)
string(FIND "${tail}" "\n}" finish)
math(EXPR length "${finish} + 2")
string(SUBSTRING "${tail}" 0 ${length} lunge)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/sword_under_test.h" "${code}\n${lunge}\n")
add_executable(sword_regression sword_regression.c ../../engine/code/game/bg_pmove.c ../../engine/code/game/bg_slidemove.c ../../engine/code/game/bg_misc.c
 ../../engine/code/qcommon/q_math.c ../../engine/code/qcommon/q_shared.c)
target_compile_definitions(sword_regression PRIVATE _CRT_SECURE_NO_WARNINGS)
target_include_directories(sword_regression PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(NOT MSVC)
 target_link_libraries(sword_regression PRIVATE m)
endif()
foreach(case IN ITEMS reach-still reach-fast bot-reach wall-hit wall-near deduplicate corpse target-wall target-team lunge-mirror lunge-wall lunge-bot lunge-close target-dead combo min-range chain-wall chain-visible lunge-corpse corpse-shield chain-corpse lunge-far target-cone swing-normal swing-slow buffer-normal buffer-slow)
 add_test(NAME sword.${case} COMMAND sword_regression ${case})
endforeach()
