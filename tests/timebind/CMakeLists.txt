cmake_minimum_required(VERSION 3.25)
project(timebind_regression C)
add_executable(timebind_regression timebind_regression.c ../../engine/code/qcommon/q_shared.c)
target_compile_definitions(timebind_regression PRIVATE _CRT_SECURE_NO_WARNINGS)
# Compile the actual two clock functions without linking every gameplay system.
# Reconfigure whenever their source changes; there is no copied implementation.
set(clock_source "${CMAKE_CURRENT_SOURCE_DIR}/../../engine/code/game/g_active.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${clock_source}")
file(READ "${clock_source}" source)
string(FIND "${source}" "static void G_UpdateTimeBind(" begin)
string(FIND "${source}" "void G_ReleaseTimeBind(" release)
if(begin LESS 0 OR release LESS begin)
  message(FATAL_ERROR "Clock function boundaries changed; update the harness")
endif()
string(SUBSTRING "${source}" ${release} -1 tail)
string(FIND "${tail}" "\n}" finish)
if(finish LESS 0)
  message(FATAL_ERROR "Cannot find clock release function end")
endif()
math(EXPR length "${release} - ${begin} + ${finish} + 2")
string(SUBSTRING "${source}" ${begin} ${length} clock_code)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/clock_under_test.h" "${clock_code}\n")
target_include_directories(timebind_regression PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(NOT MSVC)
  target_link_libraries(timebind_regression PRIVATE m)
endif()
