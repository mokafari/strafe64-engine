cmake_minimum_required(VERSION 3.25)
project(video_modes_regression C)
set(video_source "${CMAKE_CURRENT_SOURCE_DIR}/../../engine/code/q3_ui/ui_video.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${video_source}")
file(READ "${video_source}" source)
set(extracted "")
foreach(section IN ITEMS tables enumeration)
  if(section STREQUAL "tables")
    set(first "static const char *builtinResolutions")
    set(last "static void GraphicsOptions_GetInitialVideo")
  else()
    set(first "static void GraphicsOptions_GetResolutions")
    set(last "static void GraphicsOptions_CheckConfig")
  endif()
  string(FIND "${source}" "${first}" begin)
  string(FIND "${source}" "${last}" end)
  if(begin LESS 0 OR end LESS begin)
    message(FATAL_ERROR "Video mode extraction boundaries changed")
  endif()
  math(EXPR length "${end} - ${begin}")
  string(SUBSTRING "${source}" ${begin} ${length} part)
  string(APPEND extracted "${part}\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/video_under_test.h" "${extracted}")
add_executable(video_modes_regression video_modes_regression.c ../../engine/code/qcommon/q_shared.c)
target_compile_definitions(video_modes_regression PRIVATE _CRT_SECURE_NO_WARNINGS)
target_include_directories(video_modes_regression PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME video.resolution_selection COMMAND video_modes_regression)
